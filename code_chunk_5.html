```{sql}
--| connection: db1
--| output.var: disaster_brics
SELECT
    Country,
    "Start Year",
    SUM("Total Deaths") AS total_deaths
FROM natural_disaster
WHERE Country IN (
    'Brazil',
    'India',
    'China',
    'South Africa',
    'Russian Federation'
)
GROUP BY
    Country, "Start Year"
ORDER BY
    Country, "Start Year";
```
