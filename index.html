#| context: setup
#| message: false

disaster_data <- readxl::read_xlsx("public_emdat.xlsx")

    db1,
    "natural_disaster",

SQL chunk to filter data and save as R dataframe

```{sql}
--| connection: db1
--| output.var: disaster_brics
SELECT
    Country,
    "Start Year",
    SUM("Total Deaths") AS total_deaths
FROM natural_disaster
WHERE Country IN (
    'Brazil',
    'India',
    'China',
    'South Africa',
    'Russian Federation'
)
GROUP BY
    Country, "Start Year"
ORDER BY
    Country, "Start Year";
```

GGSQL chunk for scatter plot using R dataframe

```{ggsql}
#| echo: false
SELECT
    LN("Total Damage ('000 US$)" + 1) AS cost,
    LN("Total Affected" + 1) AS affected,
    "Start Year",
    "Total Affected",
    "Total Deaths",
    "Country",
    "Disaster Subgroup"
FROM r:disaster_data
WHERE "Country"IN (
    'Brazil',
    'India',
    'China',
    'South Africa',
    'Russian Federation'
)
AND"Start Year"IN (
    2020,
    2021,
    2022,
    2023,
    2024
)
VISUALIZE
    cost AS x,
    affected AS y
DRAW point
MAPPING
    Country AS stroke,
    Country AS fill
SETTING
    size => 6
DRAW text
MAPPING
    "Disaster Subgroup" AS label
SETTING
    fontsize => 4,
    vjust => 1
LABEL
    title => 'Log(Total Damage) vs Log(Total Affected)'
```
